function debounce(fn, wait) {
  let t;
  return (...args) => {
    clearTimeout(t);                          // every call wipes the pending timer...
    t = setTimeout(() => fn(...args), wait);  // ...and starts a fresh one
  };
}

function throttle(fn, wait) {
  let waiting = false;
  return (...args) => {
    if (waiting) return;                      // in the cooldown? drop it
    fn(...args);                              // otherwise fire straight away...
    waiting = true;
    setTimeout(() => { waiting = false; }, wait);  // ...then reopen after `wait`
  };
}
